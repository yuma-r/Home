    void Update()
    {
        if (autoMoving && !InUi)
        {
	    //Mantle
	    //Moves the player along Curve
            Teleport(Vector3.MoveTowards(transform.position, MoveCurve[curveIndex], Time.deltaTime * 5));
            PlayerHeight = Vector3.MoveTowards(Vector3.one * PlayerHeight, Vector3.one * Vector3.Distance(MoveCurve[curveIndex], CameraCurve[curveIndex]), Time.deltaTime * 5).x;
            //PlayerHeight = Vector3.Distance(MoveCurve[curveIndex], CameraCurve[curveIndex]);
            if (Vector3.Distance(transform.position, MoveCurve[curveIndex]) < 0.1f)
            {
		//Next Curve point
                curveIndex++;
            }
            if(curveIndex >= MoveCurve.Count)
            {
		//arriving at target
                autoMoving = false;
                Teleport(autoMovingTarget);
            }
        }
		
	//Reset sliding velocity
        if (!isSliding && slidingDir != Vector3.zero)
        {
            slidingDir = Vector3.zero;
        }
		
	//Applying values
        PlayerHeight = Mathf.Clamp(PlayerHeight, 0.9f, 1.8f);
        Camera.main.transform.parent.position = transform.position + Vector3.up * (PlayerHeight - 0.1f);

	//set player pos for npc
        PlayerPos = transform.position;
    }

    //Enabels auto move 
    public void MoveTo(Vector3 position, float targetHight, float globalCelingHight, float globalGapHight)
    {
        curveIndex = 0;
        autoMoving = true;
        autoMovingTarget = position;
        autoMovingStart = transform.position;

        Vector3 center = GetCenter(autoMovingStart, autoMovingTarget);
        Vector3 centerPoint = new Vector3(center.x, autoMovingTarget.y + GetDistanceTop(autoMovingStart, autoMovingTarget) / 2, center.z);
        MoveCurve = GetQuadraticBezierPoints(autoMovingStart, centerPoint, autoMovingTarget, 10);

        float centerHeight = globalCelingHight + 0.15f;

        Vector3 centerC = GetCenter(autoMovingStart + Vector3.up * 1.7f, autoMovingTarget + Vector3.up * (targetHight - 0.1f));
        Vector3 centerPointC = new Vector3(centerC.x, centerHeight, centerC.z);
        //Vector3 centerPointC = new Vector3(center.x, (autoMovingTarget + Vector3.up * targetHight).y + GetDistanceTop(autoMovingStart + Vector3.up * 1.7f, autoMovingTarget + Vector3.up * targetHight) / 2, center.z);
        CameraCurve = GetQuadraticBezierPoints(autoMovingStart + Vector3.up * 1.7f, centerPointC, autoMovingTarget + Vector3.up * (targetHight - 0.1f), 10);
    }

    //Tests if ledge is climabel
    public JumpGap GetWallHight(float minHight, float maxDistance, bool debug)
    {
        float celingHight = 2.5f;

        RaycastHit hit;
        if (Physics.Raycast(transform.position, Vector3.up, out hit, 2.5f, GeometryLayer))
        {
            celingHight = hit.distance;
            if (debug)
                Debug.DrawLine(transform.position, hit.point);
        }
        else if (debug)
        {
            Debug.DrawLine(transform.position, transform.position + Vector3.up * 2.5f);
        }

        float wallHight = -1;
        if(celingHight > 1)
        {
            float gapSize = 0;
            float wallDistance = 0;
            for (int i = 1; i - 1 < Mathf.Floor(celingHight * 10); i++)
            {
                RaycastHit h;
                if (Physics.Raycast(transform.position + Vector3.up * i / 10, transform.forward, out h, maxDistance, GeometryLayer))
                {
                    if (debug)
                        Debug.DrawRay(transform.position + Vector3.up * i / 10, transform.forward * (maxDistance * 0.5f), Color.red, 2);
                    if(gapSize > 5 && wallHight > minHight + 0.01f)
                    {
                        //print(wallHight);
                        return new JumpGap(wallHight, gapSize / 10f, wallDistance, transform.position.y + wallHight);
                    }
                    else if (gapSize > -1)
                    {
                        //print(gapSize);
                        gapSize = -1;
                        wallHight = -1;
                        wallDistance = h.distance;
                    }
                }
                else
                {
                    if (gapSize < 0)
                    {
                        if (debug)
                            Debug.DrawRay(transform.position + Vector3.up * i / 10, transform.forward * (maxDistance * 0.5f), Color.green, 2);
                        wallHight = i / 10f;
                        //print(wallHight);
                    }
                    else if (gapSize > 5 && wallHight > minHight + 0.01f && debug)
                    {
                        Debug.DrawRay(transform.position + Vector3.up * i / 10, transform.forward * maxDistance, Color.cyan, 2);
                        Debug.DrawRay(transform.position + Vector3.up * wallHight + transform.forward * (maxDistance * 0.5f), transform.forward * (maxDistance * 0.25f), Color.blue, 2);
                    }
                    else if (debug)
                    {
                        Debug.DrawRay(transform.position + Vector3.up * i / 10, transform.forward * (maxDistance * 0.25f), Color.white, 2);
                    }
                    gapSize++;
                }
            }
            if (gapSize > 5 && wallHight > minHight + 0.01f)
            {
                //print(wallHight);
                return new JumpGap(wallHight, gapSize / 10f, wallDistance, transform.position.y + wallHight);
            }
        }
        return null;
    }

    //Created Curve
    List<Vector3> GetQuadraticBezierPoints(Vector3 p0, Vector3 p1, Vector3 p2, int count)
    {
        List<Vector3> points = new List<Vector3>();
        for (int i = 0; i < count; i++)
        {
            float t = i / (float)(count - 1);
            Vector3 point = Mathf.Pow(1 - t, 2) * p0 +
                            2 * (1 - t) * t * p1 +
                            Mathf.Pow(t, 2) * p2;
            points.Add(point);
        }
        return points;
    }
}

public class JumpGap
{
    public float wallHight = -1;
    public float wallDistance = -1;
    public float gapsize = -1;
    public float globalWallHight = -1;
    public JumpGap(float wallHight, float gapsize, float wallDistance)
    {
        this.wallHight = wallHight;
        this.wallDistance = wallDistance;
        this.gapsize = gapsize;
        Player p = GameObject.FindFirstObjectByType<Player>();
        if(p != null)
        {
            globalWallHight = p.transform.position.y + wallHight;
        }
    }
    public JumpGap(float wallHight, float gapsize, float wallDistance, float globalWallHight)
    {
        this.wallHight = wallHight;
        this.wallDistance = wallDistance;
        this.gapsize = gapsize;
        this.globalWallHight = globalWallHight;
    }
}
